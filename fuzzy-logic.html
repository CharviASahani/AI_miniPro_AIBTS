Implementation

The mathematics behind the magic.

The core implementation below uses triangular and trapezoidal membership functions, Mamdani-style rule evaluation and centroid defuzzification.

fuzzy-logic.js 
function triangle(x, a, b, c) {
if (x <= a || x >= c) return 0;
if (x === b) return 1;

if (x < b) {
    return (x - a) / (b - a);
}

return (c - x) / (c - b);


}

function trapezoid(x, a, b, c, d) { if (x <= a || x >= d) return 0; if (x >= b && x <= c) return 1;


if (x < b) {
    return (x - a) / (b - a);
}

return (d - x) / (d - c);


}
